cmake_minimum_required(VERSION 3.5)
project(nso_scout_radius_driver_v1)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs std_msgs)
catkin_package(CATKIN_DEPENDS roscpp geometry_msgs std_msgs)
add_executable(nso_scout_attended_node radius_driver_node.cpp)
target_include_directories(nso_scout_attended_node PRIVATE . ${catkin_INCLUDE_DIRS} ${NSO_UGV_SDK_INCLUDE_DIR})
target_link_libraries(nso_scout_attended_node ${NSO_UGV_SDK_LIBRARY} ${catkin_LIBRARIES} Threads::Threads)
target_compile_options(nso_scout_attended_node PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(attended_gate_test attended_gate_test.cpp)
target_link_libraries(attended_gate_test Threads::Threads)
target_compile_options(attended_gate_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME attended_gate COMMAND attended_gate_test)
add_executable(timing_test timing_test.cpp)
target_link_libraries(timing_test Threads::Threads)
target_compile_options(timing_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME timing_test COMMAND timing_test)
